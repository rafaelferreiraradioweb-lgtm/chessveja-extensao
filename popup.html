document.getElementById('btn-analisar').addEventListener('click', async () => { const resultDiv = document.getElementById('result'); resultDiv.innerHTML = "Analisando a partida no nível Grande Mestre... ⏳
Aguarde, extraindo momentos críticos."; // Busca a aba do Lichess (funciona perfeitamente no Painel Lateral) const [tab] = await chrome.tabs.query({ active: true, lastFocusedWindow: true }); if (!tab || !tab.url || !tab.url.includes("lichess.org")) { resultDiv.innerHTML = "Erro: Você precisa estar na aba do Lichess."; return; } // Extrai o ID da partida diretamente do link (ignorando páginas como 'analysis') const urlMatch = tab.url.match(/lichess\.org\/([a-zA-Z0-9]{8})/); const palavrasInvalidas = ['analysis', 'training', 'practice', 'study']; let gameId = null; if (urlMatch && !palavrasInvalidas.includes(urlMatch[1])) { gameId = urlMatch[1]; } if (!gameId) { resultDiv.innerHTML = "Erro: Por favor, abra uma partida finalizada no Lichess."; return; } try { const pgnRes = await fetch(`https://lichess.org/game/export/${gameId}`); if (!pgnRes.ok) throw new Error("Partida não encontrada"); const pgn = await pgnRes.text(); const apiRes = await fetch('https://chessveja-site.vercel.app/api/analyze', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ pgn, level: document.getElementById('level').value, type: 'summary' }) }); const data = await apiRes.json(); if (data.analysis) { let textoFormatado = data.analysis; textoFormatado = textoFormatado.replace(/\n/g, '
'); textoFormatado = textoFormatado.replace(/\*\*(.*?)\*\*/g, '$1'); textoFormatado = textoFormatado.replace(/\[(.*?)\]\((.*?)\)/g, '

📱 $1'); resultDiv.innerHTML = textoFormatado; } else { resultDiv.innerHTML = "Erro na análise. Tente novamente."; } } catch (err) { resultDiv.innerHTML = "Erro de conexão. Verifique sua internet ou tente mais tarde."; } });
